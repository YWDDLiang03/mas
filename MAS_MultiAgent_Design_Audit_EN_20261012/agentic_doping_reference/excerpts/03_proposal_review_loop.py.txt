        while True:
            dialogue.append({"role": "proposer", "round": round_,
                             "dopants": [d["dopant"] for d in current["dopants"]],
                             "rationale": current.get("rationale", "")})
            ta = harness.step_type_a(current["dopants"], known)
            _write_versioned(run_id, "proposal_agent", slot, f"PROPOSAL_v{round_}.json", {
                "proposal_id": slot, "round": round_, "parent": parent_formula, "site_type": site_type,
                "host_species": host_species, "dopants": current["dopants"],
                "rationale": current.get("rationale", ""), "revision_addressing": current.get("revision_addressing", {}),
                "type_a_selfcheck": ta, "proposer_method": init["method"], "at": now_iso()})
            if not ta["passed"]:
                if round_ < cfg.max_rounds:
                    rev = harness.step_revise(current["dopants"], ta["reasons"], dialogue=dialogue)
                    current = rev["proposal"]
                    round_ += 1
                    continue
                slot_kill = "; ".join(ta["reasons"]) or "type-a self-check failed"
                _append_failure(run_id, known, parent_formula, site_type, current["dopants"], slot_kill)
                status = "killed"
                break

            review = harness.step_review(current["dopants"], dialogue=dialogue)
            same_family = same_family or review.get("same_family_warning")
            _write_versioned(run_id, "critic_agent", slot, f"REVIEW_v{round_}.json", {
                "proposal_id": slot, "round": round_, **review, "at": now_iso()})
            comb = review["combined"]
            slot_score = comb.get("score")
            history.append({"round": round_, "dopants": [d["dopant"] for d in current["dopants"]],
                            "verdict": comb["verdict"], "score": comb.get("score"),
                            "promise_score": comb.get("promise_score"),
                            "evidence_confidence": comb.get("evidence_confidence"),
                            "evidence_basis": comb.get("evidence_basis"),
                            "evidence_refs": comb.get("evidence_refs", []),
                            "raw_evidence_confidence": comb.get("raw_evidence_confidence"),
                            "raw_evidence_basis": comb.get("raw_evidence_basis"),
                            "raw_evidence_refs": comb.get("raw_evidence_refs", []),
                            "evidence_audit": comb.get("evidence_audit"),
                            "evaluation_worthy": comb.get("evaluation_worthy"),
                            "high_promise": comb.get("high_promise"), "by": comb.get("by")})
            dialogue.append({"round": round_, "role": "reviewer", "verdict": comb["verdict"],
                             "score": comb.get("score"), "promise_score": comb.get("promise_score"),
                             "evidence_confidence": comb.get("evidence_confidence"),
                             "evidence_basis": comb.get("evidence_basis"),
                             "evidence_refs": comb.get("evidence_refs", []),
                             "evaluation_worthy": comb.get("evaluation_worthy"),
                             "high_promise": comb.get("high_promise"),
                             "concerns": review.get("concerns", [])})
            # Reviewer concerns remain in the versioned run trace.  They are opinions about this
            # candidate, not durable scientific facts, so they must not create cross-run lessons or
            # auto-promoted skills. Deterministic hard gates are enforced directly by rule_review.

            if comb["accepted"]:
                accepted.append(current["dopants"])
                accepted_details.append({"dopants": current["dopants"], "review": review,
                                         "slot": slot, "status": "completed"})
                harness.note_acceptance()      # credit the skills that informed this proposal
                status = "completed"
                break
            if comb["verdict"] == "reject":
                slot_kill = comb.get("kill_reason")
                _append_failure(run_id, known, parent_formula, site_type, current["dopants"],
                                comb.get("kill_reason") or "rejected")
                run_dir.write_json_atomic(
                    run_dir.run_path(run_id, "critic_agent", "proposals", slot, "KILL_ARGUMENT.json"),
                    {"proposal_id": slot, "round": round_, "killed_by": comb.get("by"),
                     "kill_reason": comb.get("kill_reason"), "dopants": current["dopants"], "at": now_iso()})
                status = "killed"
                break
            if round_ < cfg.max_rounds:
                rev = harness.step_revise(current["dopants"], review["concerns"], dialogue=dialogue)
                current = rev["proposal"]
                round_ += 1
                continue
            # Max rounds without a quality-threshold accept. A hard failure is killed; a merely
            # testable-but-low-promise proposal remains auditable as exhausted but is not admitted to
            # the recommendation plan. Treating exhausted as accepted made the reviewer gate cosmetic.
            if comb.get("llm_rejected"):
                slot_kill = (comb.get("kill_reason")
                             or "exhausted after repeated non-viable picks (reviewer kept rejecting)")
                _append_failure(run_id, known, parent_formula, site_type, current["dopants"], slot_kill)
                status = "killed"
                break
            status = "exhausted"
            break

        run_dir.write_json_atomic(
            run_dir.run_path(run_id, "critic_agent", "proposals", slot, "REVIEW_STATE.json"),
            {"proposal_id": slot, "status": status, "rounds": round_, "history": history,
             "final_dopants": current["dopants"], "at": now_iso()})
        slots.append({"slot": slot, "status": status, "rounds": round_,
                      "dopants": [d["dopant"] for d in current["dopants"]],
                      "accepted": status == "completed"})
        # Keep the per-slot result in the auditable run artifacts above. A reviewer outcome by itself
        # is not evidence and therefore is not promoted into inherited memory.
        try:
            mem.record_outcome(parent_formula, host_species, site_type, current["dopants"],
                               status, reason=slot_kill, score=slot_score)
