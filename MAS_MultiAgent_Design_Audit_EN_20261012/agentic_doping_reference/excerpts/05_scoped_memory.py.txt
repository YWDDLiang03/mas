    def _recallable(self, rec: dict) -> bool:
        """Approved memory is inherited; provisional memory is same-group/self-learning only."""
        if _memory_ready(rec):
            return True
        if _review_state(rec) != "provisional" or self.scope == SUPER:
            return False
        if str(rec.get("group_id") or "default") != str(self.group_id):
            return False
        origin_scope = str(rec.get("mem_scope") or "public")
        # Group-public provisional notes may be tried by both agents in this group. Private notes
        # remain visible only to the agent that authored them. Neither form can cross groups.
        return origin_scope == "public" or origin_scope == self.scope
