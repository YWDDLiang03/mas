    ids = _clean_terms(candidate_ids)
    if any(_contains_term(combined, value) for value in ids):
        violations.append("candidate_identifier")

    material_terms = _clean_terms(materials)
    if any(_contains_term(combined, value) for value in material_terms) or _contains_specific_formula(combined):
        violations.append("specific_material_or_dopant_set")

    mentioned_dopants = [value for value in _clean_terms(dopants)
                         if _contains_term(combined, value, case_sensitive=True)]
    if len(mentioned_dopants) >= 2:
        violations.append("specific_material_or_dopant_set")
    elif mentioned_dopants and _NUMBER_RE.search(combined):
        violations.append("candidate_specific_parameters")

    if _SCORE_WORD_RE.search(combined) and _NUMBER_RE.search(combined):
        violations.append("observed_score_or_threshold")
    if _PROPERTY_WORD_RE.search(combined) and _NUMBER_RE.search(combined):
        violations.append("observed_performance_value")

    scientific, source = _has_asserted_conclusion(combined)
    if scientific:
        violations.append("performance_conclusion")
    if source:
        violations.append("literature_or_experiment_conclusion")

    if fields["when"] and fields["procedure"] and not _METHOD_ACTION_RE.search(
        fields["when"] + "\n" + fields["procedure"]
    ):
        violations.append("not_reusable_methodology")
