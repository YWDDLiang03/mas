def discover(goal: str, gaps: list, *, site_type: Optional[str] = None, group_id: str = "default",
             run_id: Optional[str] = None, override: Optional[dict] = None,
             lit_override: Optional[dict] = None, depth: str = "low") -> dict:
    """For each gap: targeted literature search (reuses research_for_goal: fetch+screen+read+cache),
    then distil the genuinely-NEW papers into knowledge. Returns a summary of what was discovered."""
    from agentic_doping.literature import research_for_goal
    from agentic_doping.literature.visibility import resolve_papers
    from agentic_doping.literature.distill import distill_to_knowledge
    from agentic_doping.literature.quality import (distillable_literature_claims,
                                                   knowledge_is_injectable)
    from agentic_doping.harness.memory import MemoryStore
    km = MemoryStore("public", group_id=group_id, run_id=run_id)   # group public (super_public is human-only)
    # Pending/legacy noise is retained in JSONL for audit, but cannot make a paper look as though it
    # has already crossed the quality boundary.  A future better read may therefore repair it.
    existing = [k for k in km.knowledge() if knowledge_is_injectable(k)]
    distilled = {g.get("cache_key") for k in existing for g in (k.get("grounding") or []) if g.get("cache_key")}
    out = {"gaps": list(gaps), "new_papers": 0, "new_knowledge": 0, "per_gap": []}
    for gap in gaps:
        try:
            rep = research_for_goal(gap, site_type=site_type, override=(lit_override or override),
                                    depth=depth, group_id=group_id, run_id=run_id)
        except Exception:
            out["per_gap"].append({"gap": gap, "new_papers": 0, "new_knowledge": 0, "error": "fetch_failed"})
            continue
        papers = resolve_papers(rep)
        fresh_keys = {p.get("key") for p in papers if p.get("key") and p["key"] not in distilled}
        if not fresh_keys:
            out["per_gap"].append({"gap": gap, "new_papers": 0, "new_knowledge": 0})
            continue
        claims = distillable_literature_claims(papers, fresh_keys)
        nk = 0
        for e in (distill_to_knowledge(gap, claims, papers, existing, override=(lit_override or override)) if claims else []):
            if km.add_knowledge(e.get("statement", ""), body=e.get("body", ""), tags=e.get("tags"),
                                parent=e.get("parent"), site=e.get("site"), dopants=e.get("dopants"),
                                property=e.get("property"), effect_direction=e.get("effect_direction"),
                                ratio=e.get("ratio"), evidence_scope=e.get("evidence_scope"),
                                evidence_tier=e.get("evidence_tier"),
                                evidence_confidence=e.get("evidence_confidence"),
                                quality_status=e.get("quality_status"),
                                review_state=e.get("review_state"), evidence_kind=e.get("evidence_kind"),
                                confidence=e.get("confidence"), status=e.get("status", "open"),
                                grounding=e.get("grounding"), source="gap_discovery", run_id=run_id):
                nk += 1
                existing.append(e)
        distilled |= fresh_keys
        out["new_papers"] += len(fresh_keys)
        out["new_knowledge"] += nk
        out["per_gap"].append({"gap": gap, "new_papers": len(fresh_keys), "new_knowledge": nk})
    return out


def deepen(goal: str, *, site_type: Optional[str] = None, group_id: str = "default",
           run_id: Optional[str] = None, override: Optional[dict] = None,
           lit_override: Optional[dict] = None, depth: str = "low", max_gaps: int = 4) -> dict:
    """Full pass: read knowledge+lessons -> identify_gaps -> discover. LIGHT inline uses max_gaps=1,
    depth='low'; the DEEP on-demand pass uses more gaps + higher depth."""
    from agentic_doping.harness.memory import MemoryStore
    km = MemoryStore("public", group_id=group_id)   # read the group's knowledge (super_public is human-only)
    lm = MemoryStore("public", group_id=group_id)
    knowledge = km.relevant_knowledge(site=site_type, goal=goal, limit=30)
    lessons = lm.relevant(site=site_type, goal=goal, limit=20).get("lessons", [])
    gaps = identify_gaps(goal, knowledge, lessons, override=override, k=max_gaps)
    if not gaps:
        return {"gaps": [], "new_papers": 0, "new_knowledge": 0, "per_gap": []}
    return discover(goal, gaps, site_type=site_type, group_id=group_id, run_id=run_id,
                    override=override, lit_override=lit_override, depth=depth)
