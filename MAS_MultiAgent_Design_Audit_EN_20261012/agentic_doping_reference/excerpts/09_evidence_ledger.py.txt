class Ledger:
    def __init__(self, run_id: str):
        self.run_id = run_id
        self.path: Path = run_path(run_id, "ledger", LEDGER_NAME)

    def append(self, record: LedgerRecord) -> None:
        append_jsonl(self.path, record.model_dump(mode="json"))

    def load(self) -> list[LedgerRecord]:
        if not self.path.exists():
            return []
        records: list[LedgerRecord] = []
        for line in self.path.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if not line:
                continue
            try:
                records.append(LedgerRecord.model_validate_json(line))
            except Exception:
                # Skip malformed/partial lines (crash recovery) rather than abort.
                continue
        return records

    def latest_by_candidate(self) -> dict[str, LedgerRecord]:
        latest: dict[str, LedgerRecord] = {}
        for rec in self.load():
            cur = latest.get(rec.candidate_id)
            if cur is None or rec.updated_at >= cur.updated_at:
                latest[rec.candidate_id] = rec
