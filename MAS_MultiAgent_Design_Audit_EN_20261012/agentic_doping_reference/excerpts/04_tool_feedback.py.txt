        kwargs = {"model": cfg.model_name, "max_tokens": cur_max, "temperature": temperature,
                  "messages": convo, "tools": tools}
        if system:
            kwargs["system"] = system
        try:
            resp = _with_retry(lambda: client.messages.create(**kwargs))
        except Exception as exc:   # noqa: BLE001
            raise LLMUnavailable(f"{cfg.provider} agent loop error: {type(exc).__name__}: {exc}") from exc
        stop = getattr(resp, "stop_reason", None)
        txt = "".join(getattr(b, "text", "") for b in resp.content if getattr(b, "type", None) == "text").strip()
        if txt:
            final_text = txt
        # truncation escalation: bump and RETRY the same turn (don't consume the result)
        if stop == "max_tokens" and cur_max < _AGENT_LOOP_MAX_TOKENS:
            cur_max = min(_AGENT_LOOP_MAX_TOKENS, cur_max * 2)
            it += 1
            continue
        # rebuild assistant content as plain dicts (text + tool_use only; drop provider 'thinking'
        # blocks which would need a signature to echo back)
        assistant_content = []
        for b in resp.content:
            t = getattr(b, "type", None)
            if t == "text":
                assistant_content.append({"type": "text", "text": b.text})
            elif t == "tool_use":
                assistant_content.append({"type": "tool_use", "id": b.id, "name": b.name,
                                          "input": dict(b.input or {})})
        if assistant_content:
            convo.append({"role": "assistant", "content": assistant_content})
        tool_uses = [b for b in resp.content if getattr(b, "type", None) == "tool_use"]
        if stop != "tool_use" or not tool_uses:
            break
        results = []
        for b in tool_uses:
            try:
                out = dispatch(b.name, dict(b.input or {}))
            except Exception as exc:   # noqa: BLE001 — a tool error is fed back, not fatal
                out = {"error": f"{type(exc).__name__}: {exc}"}
            tool_calls.append({"name": b.name, "input": dict(b.input or {}), "result": out})
            results.append({"type": "tool_result", "tool_use_id": b.id,
                            "content": json.dumps(out, ensure_ascii=False, default=str)})
        convo.append({"role": "user", "content": results})
        it += 1
